const badWords = ["广告", "诈骗", "违法"];

function onEnter() {
    infomsg("欢迎加入本群，请遵守群规~");
}

function onSend(msg) {
    if (msg.content.checkIfrom(badWords)) {
        WSystemMsg("⚠️ 检测到违规内容，消息已被屏蔽");
        return false;  // 阻止发送
    }
}

安全 · 透明 · 可审计

哈希链技术让每一条消息都可追溯、可验证

哈希链原理

每条消息记录前一条消息的哈希值，形成防篡改链条。任何修改都会破坏链条完整性。

证据包导出

一键导出包含完整哈希链的 JSON 证据包，第三方可独立验证聊天记录真伪。

本地存储

所有聊天记录存储在浏览器 IndexedDB，数据归用户所有，服务端不留存。
